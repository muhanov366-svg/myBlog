/**ОСНОВНЫЕ ФУНКЦИИ:
 *  1. Регистрация и авторизация пользователей
 *  2. Создание, чтение, редактирование и удаление постов
 *  3. Система подписок (подписаться/отписаться)
 *  4. Комментарии к постам
 *  5. Разграничение доступа к приватным постам
 * 
 *  СТРУКТУРА ТАБЛИЦ:
 *  - Users: ID, Имя, Email, Пароль, Подписки
 *  - Posts: ID, UserID, Заголовок, Текст, Теги, Приватность, Дата, Разрешенные пользователи
 *  - Comments: ID, PostID, UserID, Текст, Дата
 */
function doGet(e) {
  // Оборачиваем весь код в try-catch для перехвата любых ошибок
  try {
    // Извлекаем параметры из запроса
    // e - объект запроса, содержит все переданные параметры
    const params = e.parameter;
    // action - определяет, какое действие нужно выполнить
    const action = params.action;
    // Все остальные данные хранятся в params
    const data = params;
    
    // Переменная для хранения результата выполнения
    let result;
    
    // Используем switch для выбора нужного обработчика в зависимости от action
    switch(action) {
      // Регистрация нового пользователя
      case 'register': 
        result = handleRegister(data); 
        break;
      
      // Вход в систему (логин)
      case 'login': 
        result = handleLogin(data); 
        break;
      
      // Добавление нового поста
      case 'addPost': 
        result = handleAddPost(data); 
        break;
      
      // Получение списка постов
      case 'getPosts': 
        result = handleGetPosts(data); 
        break;
      
      // Редактирование существующего поста
      case 'editPost': 
        result = handleEditPost(data); 
        break;
      
      // Удаление поста
      case 'deletePost': 
        result = handleDeletePost(data); 
        break;
      
      // Подписка на пользователя
      case 'subscribe': 
        result = handleSubscribe(data); 
        break;
      
      // Отписка от пользователя
      case 'unsubscribe': 
        result = handleUnsubscribe(data); 
        break;
      
      // Добавление комментария к посту
      case 'addComment': 
        result = handleAddComment(data); 
        break;
      
      // Получение комментариев к посту
      case 'getComments': 
        result = handleGetComments(data); 
        break;
      
      // Получение информации о пользователе по ID
      case 'getUserById': 
        result = handleGetUserById(data); 
        break;
      
      // Тестовый эндпоинт для проверки работоспособности сервера
      case 'test': 
        result = { success: true, message: '✅ Сервер работает!' }; 
        break;
      
      // Если action не распознан, возвращаем ошибку
      default: 
        result = { success: false, error: '❌ Неизвестное действие' };
    }
    
    /**
     * Формируем ответ для клиента
     * ContentService - специальный сервис Google Apps Script для работы с контентом
     * createTextOutput - создает текстовый ответ
     * setMimeType - устанавливает тип контента как JSON
     */
    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    /**
     * Если произошла ошибка, возвращаем ее клиенту
     */
    return ContentService
      .createTextOutput(JSON.stringify({ 
        success: false, 
        error: error.message 
      }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ПОЛУЧЕНИЕ ТАБЛИЦЫ
function getSheet(name) {
  // getActiveSpreadsheet() - получает текущую открытую таблицу
  // getSheetByName(name) - ищет таблицу по имени
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
}

// ГЕНЕРАЦИЯ НОВОГО ID
function getNextId(sheet) {
  // Получаем номер последней заполненной строки
  let lastRow = sheet.getLastRow();
  
  // Если в таблице нет данных (только заголовки или пусто)
  if (lastRow < 1) return 1;
  
  // Берем значение из последней строки первой колонки
  const lastId = sheet.getRange(lastRow, 1).getValue();
  
  // Если ID - число, просто увеличиваем на 1
  if (typeof lastId === 'number') {
    return lastId + 1;
  }
  
  // Если ID - строка, пытаемся найти число в конце
  if (typeof lastId === 'string') {
    // Регулярное выражение ищет цифры в конце строки
    const match = lastId.match(/\d+$/);
    if (match) {
      // Преобразуем найденные цифры в число и увеличиваем на 1
      return parseInt(match[0]) + 1;
    }
  }
  
  // Запасной вариант: используем номер строки
  return lastRow + 1;
}

// ПОИСК ПОЛЬЗОВАТЕЛЯ ПО EMAIL
function getUserByEmail(email) {
  // Получаем таблицу Users
  const sheet = getSheet('Users');
  if (!sheet) return null; // Если таблицы нет, возвращаем null
  
  // Получаем все данные из таблицы (двумерный массив)
  const data = sheet.getDataRange().getValues();
  
  // Приводим email к единому формату (нижний регистр, без пробелов)
  const searchEmail = String(email).trim().toLowerCase();
  
  // Проходим по всем строкам, начиная с 1 (пропускаем заголовок)
  for (let i = 1; i < data.length; i++) {
    // Получаем email из третьей колонки (индекс 2)
    const userEmail = String(data[i][2] || '').trim().toLowerCase();
    
    // Если email совпадает, создаем объект пользователя
    if (userEmail === searchEmail) {
      return {
        id: String(data[i][0]),          // ID пользователя
        name: String(data[i][1] || ''),  // Имя
        email: String(data[i][2] || ''), // Email
        password: String(data[i][3] || ''), // Пароль (в реальном проекте нужно хранить в зашифрованном виде!)
        subscriptions: String(data[i][4] || '') // Подписки (список ID через запятую)
      };
    }
  }
  return null; // Пользователь не найден
}

// ПОИСК ПОЛЬЗОВАТЕЛЯ ПО ID
function getUserById(id) {
  const sheet = getSheet('Users');
  if (!sheet) return null;
  
  const data = sheet.getDataRange().getValues();
  const searchId = String(id); // Приводим к строке для сравнения
  
  for (let i = 1; i < data.length; i++) {
    // Сравниваем ID как строки (чтобы избежать проблем с типами)
    if (String(data[i][0]) === searchId) {
      return {
        id: String(data[i][0]),          // ID
        name: String(data[i][1] || ''),  // Имя
        email: String(data[i][2] || ''), // Email
        subscriptions: String(data[i][4] || '') // Подписки
      };
    }
  }
  return null;
}

// ОБНОВЛЕНИЕ ПОДПИСОК ПОЛЬЗОВАТЕЛЯ
function updateUserSubscriptions(userId, subsArray) {
  const sheet = getSheet('Users');
  const data = sheet.getDataRange().getValues();
  
  // Ищем пользователя по ID
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(userId)) {
      // Объединяем массив в строку через запятую и записываем в 5-ю колонку
      sheet.getRange(i + 1, 5).setValue(subsArray.join(','));
      return true;
    }
  }
  return false; // Пользователь не найден
}

// РЕГИСТРАЦИЯ НОВОГО ПОЛЬЗОВАТЕЛЯ
function handleRegister(data) {
  // Деструктуризация - извлекаем нужные поля из объекта
  const { name, email, password } = data;
  
  // Проверка: все поля должны быть заполнены
  if (!name || !email || !password) {
    return { success: false, error: 'Все поля обязательны для заполнения' };
  }
  
  // Проверяем существование таблицы Users
  const sheet = getSheet('Users');
  if (!sheet) {
    return { success: false, error: 'Таблица Users не найдена' };
  }
  
  // Проверяем, не занят ли email (уникальность)
  if (getUserByEmail(email)) {
    return { success: false, error: 'Пользователь с такой почтой уже существует' };
  }

  // Генерируем новый ID
  const newId = getNextId(sheet);
   
  // Возвращаем успешный результат с данными нового пользователя
  return { 
    success: true, 
    user: { id: newId, name, email } 
  };
}

// ВХОД В СИСТЕМУ (ЛОГИН)
function handleLogin(data) {
  const { email, password } = data;
  
  // Валидация входных данных
  if (!email || !password) {
    return { success: false, error: 'Все поля обязательны для заполнения' };
  }
  
  // Поиск пользователя по email
  const user = getUserByEmail(email);
  
  // Если пользователь не найден
  if (!user) {
    return { success: false, error: 'Пользователь не найден' };
  }
  
  // Проверка пароля (в реальном приложении используйте сравнение хэшей!)
  if (user.password !== String(password)) {
    return { success: false, error: 'Неверный пароль' };
  }
  
  // Возвращаем данные пользователя (без пароля)
  return { 
    success: true, 
    user: { 
      id: user.id, 
      name: user.name, 
      email: user.email, 
      subscriptions: user.subscriptions 
    } 
  };
}

// ДОБАВЛЕНИЕ НОВОГО ПОСТА
function handleAddPost(data) {
  const { userId, title, content, tags, isPrivate, allowedUsers } = data;
  
  // Проверяем обязательные поля
  if (!userId || !title || !content) {
    return { success: false, error: 'Заголовок и текст поста обязательны' };
  }
  
  const sheet = getSheet('Posts');
  if (!sheet) {
    return { success: false, error: 'Таблица Posts не найдена' };
  }
  
  // Генерируем ID для нового поста
  const newId = getNextId(sheet);
  // Текущая дата в формате ISO (YYYY-MM-DDTHH:mm:ss.sssZ)
  const date = new Date().toISOString();
  // Приводим allowedUsers к строке (если не указан - пустая строка)
  const allowedStr = allowedUsers || '';
  
  // Добавляем пост в таблицу
  sheet.appendRow([
    newId,                         // ID поста
    userId,                        // ID автора
    title,                         // Заголовок
    content,                       // Текст
    tags || '',                   // Теги
    isPrivate === 'true' ? 'true' : 'false', // Приватность (строгое сравнение)
    date,                          // Дата создания
    allowedStr                     // Разрешенные пользователи
  ]);
  
  return { success: true, postId: newId };
}

// ПОЛУЧЕНИЕ СПИСКА ПОСТОВ
function handleGetPosts(data) {
  const sheet = getSheet('Posts');
  if (!sheet) {
    return { success: false, error: 'Таблица Posts не найдена' };
  }
  
  // Получаем все данные из таблицы
  const dataRows = sheet.getDataRange().getValues();
  const posts = [];
  
  // ID текущего пользователя для проверки доступа
  const currentUserId = data.userId;
  const onlyPrivate = data.isPrivate === 'true';
  const onlyPublic = data.isPrivate === 'false';

  // Проходим по всем строкам (начиная с 1, чтобы пропустить заголовки)
  for (let i = 1; i < dataRows.length; i++) {
    const row = dataRows[i];
    const userId = String(row[1] || '');           // ID автора
    const privateFlag = String(row[5]) === 'true'; // Приватность поста
    
    // Фильтр по приватности (если задан)
    if (onlyPrivate && !privateFlag) continue;
    if (onlyPublic && privateFlag) continue;
    
    // Проверка доступа к посту
    let canView = false;
    
    if (!privateFlag) {
      // Публичный пост - доступен всем
      canView = true;
    } else {
      // Приватный пост - доступ ограничен
      if (String(currentUserId) === userId) {
        // Автор всегда видит свой пост
        canView = true;
      } else {
        // Проверяем, есть ли пользователь в списке разрешенных
        const allowedUsersStr = String(row[7] || '');
        const allowedArray = allowedUsersStr 
          ? allowedUsersStr.split(',').map(id => String(id).trim()) 
          : [];
        
        if (allowedArray.includes(String(currentUserId))) {
          canView = true;
        }
      }
    }
    
    // Если доступ есть, добавляем пост в результат
    if (canView) {
      posts.push({
        id: String(row[0]),                        // ID поста
        userId: userId,                            // ID автора
        title: String(row[2] || ''),              // Заголовок
        content: String(row[3] || ''),            // Текст
        tags: String(row[4] || ''),               // Теги
        isPrivate: privateFlag,                   // Приватность (boolean)
        createdAt: String(row[6] || ''),          // Дата создания
        // Преобразуем строку разрешенных пользователей в массив чисел
        allowedUsers: String(row[7] || '')
          .split(',')
          .filter(id => id)
          .map(id => Number(id))
      });
    }
  }
  
  // Сортировка: сначала новые посты (по дате создания)
  posts.sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
  
  return { success: true, posts };
}

// РЕДАКТИРОВАНИЕ ПОСТА
function handleEditPost(data) {
  const { postId, userId, title, content, tags, isPrivate, allowedUsers } = data;
  
  // Проверяем наличие обязательных данных
  if (!postId || !userId) {
    return { success: false, error: 'Недостаточно данных для редактирования' };
  }
  
  const sheet = getSheet('Posts');
  if (!sheet) {
    return { success: false, error: 'Таблица Posts не найдена' };
  }
  
  const dataRows = sheet.getDataRange().getValues();
  
  // Ищем пост по ID
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === String(postId)) {
      // Проверяем, что пользователь - автор поста
      if (String(dataRows[i][1]) !== String(userId)) {
        return { success: false, error: 'Нет прав для редактирования' };
      }
      
      // Обновляем поля поста
      const allowedStr = allowedUsers || '';
      sheet.getRange(i + 1, 3).setValue(title);           // Заголовок
      sheet.getRange(i + 1, 4).setValue(content);         // Текст
      sheet.getRange(i + 1, 5).setValue(tags || '');      // Теги
      sheet.getRange(i + 1, 6).setValue(isPrivate === 'true' ? 'true' : 'false'); // Приватность
      sheet.getRange(i + 1, 8).setValue(allowedStr);      // Разрешенные пользователи
      
      return { success: true };
    }
  }
  
  return { success: false, error: 'Пост не найден' };
}

// УДАЛЕНИЕ ПОСТА
function handleDeletePost(data) {
  const { postId, userId } = data;
  
  if (!postId || !userId) {
    return { success: false, error: 'Недостаточно данных для удаления' };
  }
  
  const sheet = getSheet('Posts');
  if (!sheet) {
    return { success: false, error: 'Таблица Posts не найдена' };
  }
  
  const dataRows = sheet.getDataRange().getValues();
  
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === String(postId)) {
      // Проверяем, что пользователь - автор поста
      if (String(dataRows[i][1]) !== String(userId)) {
        return { success: false, error: 'Нет прав для удаления' };
      }
      
      // Удаляем строку с постом
      sheet.deleteRow(i + 1);
      return { success: true };
    }
  }
  
  return { success: false, error: 'Пост не найден' };
}

// ПОДПИСКА НА ПОЛЬЗОВАТЕЛЯ
function handleSubscribe(data) {
  const { currentUserId, targetUserId } = data;
  
  // Проверка наличия данных
  if (!currentUserId || !targetUserId) {
    return { success: false, error: 'Недостаточно данных для подписки' };
  }
  
  const sheet = getSheet('Users');
  if (!sheet) {
    return { success: false, error: 'Таблица Users не найдена' };
  }
  
  const dataRows = sheet.getDataRange().getValues();
  
  // Ищем текущего пользователя
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === String(currentUserId)) {
      // Получаем текущий список подписок
      let subs = String(dataRows[i][4] || '');
      let subsArray = subs ? subs.split(',').map(id => String(id).trim()) : [];
      
      // Проверяем, не подписан ли уже на этого пользователя
      if (!subsArray.includes(String(targetUserId))) {
        // Добавляем нового пользователя в список подписок
        subsArray.push(String(targetUserId));
        // Сохраняем обновленный список
        sheet.getRange(i + 1, 5).setValue(subsArray.join(','));
      }
      
      return { success: true, subscriptions: subsArray };
    }
  }
  
  return { success: false, error: 'Пользователь не найден' };
}

// ОТПИСКА ОТ ПОЛЬЗОВАТЕЛЯ
function handleUnsubscribe(data) {
  const { currentUserId, targetUserId } = data;
  
  if (!currentUserId || !targetUserId) {
    return { success: false, error: 'Недостаточно данных для отписки' };
  }
  
  const sheet = getSheet('Users');
  if (!sheet) {
    return { success: false, error: 'Таблица Users не найдена' };
  }
  
  const dataRows = sheet.getDataRange().getValues();
  
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === String(currentUserId)) {
      // Получаем текущий список подписок
      let subs = String(dataRows[i][4] || '');
      let subsArray = subs ? subs.split(',').map(id => String(id).trim()) : [];
      
      // Удаляем целевого пользователя из списка (фильтрация)
      subsArray = subsArray.filter(id => id !== String(targetUserId));
      
      // Сохраняем обновленный список
      sheet.getRange(i + 1, 5).setValue(subsArray.join(','));
      
      return { success: true, subscriptions: subsArray };
    }
  }
  
  return { success: false, error: 'Пользователь не найден' };
}

// ДОБАВЛЕНИЕ КОММЕНТАРИЯ
function handleAddComment(data) {
  const { postId, userId, text } = data;
  
  // Проверяем, что все поля заполнены
  if (!postId || !userId || !text) {
    return { success: false, error: 'Все поля комментария обязательны' };
  }
  
  const sheet = getSheet('Comments');
  if (!sheet) {
    return { success: false, error: 'Таблица Comments не найдена' };
  }
  
  // Генерируем ID для нового комментария
  const newId = getNextId(sheet);
  // Текущая дата и время
  const date = new Date().toISOString();
  
  // Добавляем комментарий в таблицу
  // Колонки: [ID, PostID, UserID, Текст, Дата]
  sheet.appendRow([newId, postId, userId, text, date]);
  
  return { success: true, commentId: newId };
}

// ПОЛУЧЕНИЕ КОММЕНТАРИЕВ ПОСТА
function handleGetComments(data) {
  const { postId } = data;
  
  if (!postId) {
    return { success: false, error: 'ID поста не указан' };
  }
  
  const sheet = getSheet('Comments');
  if (!sheet) {
    return { success: false, error: 'Таблица Comments не найдена' };
  }
  
  const dataRows = sheet.getDataRange().getValues();
  const comments = [];
  
  // Проходим по всем строкам и ищем комментарии для указанного поста
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][1]) === String(postId)) {
      comments.push({
        id: String(dataRows[i][0]),          // ID комментария
        postId: String(dataRows[i][1]),      // ID поста
        userId: String(dataRows[i][2]),      // ID автора
        text: String(dataRows[i][3] || ''), // Текст
        createdAt: String(dataRows[i][4] || '') // Дата создания
      });
    }
  }
  
  // Сортировка: сначала старые комментарии (хронологический порядок)
  comments.sort((a, b) => new Date(a.createdAt) - new Date(b.createdAt));
  
  return { success: true, comments };
}

// ПОЛУЧЕНИЕ ПОЛЬЗОВАТЕЛЯ ПО ID
function handleGetUserById(data) {
  const { userId } = data;
  
  if (!userId) {
    return { success: false, error: 'ID пользователя не указан' };
  }
  
  const user = getUserById(userId);
  
  if (!user) {
    return { success: false, error: 'Пользователь не найден' };
  }
  
  // Возвращаем данные (без пароля, т.к. getUserById его не включает)
  return { success: true, user };
}
