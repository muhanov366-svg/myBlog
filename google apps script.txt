/**
 * ОСНОВНЫЕ ФУНКЦИИ:
 *  1. Регистрация и авторизация пользователей (с хэшированием пароля и токенами сессий)
 *  2. Создание, чтение, редактирование и удаление постов
 *  3. Система подписок (подписаться/отписаться)
 *  4. Комментарии к постам
 *  5. Разграничение доступа к приватным постам + запросы доступа
 *
 *  СТРУКТУРА ТАБЛИЦ:
 *  - Users:    ID | Имя | Email | PasswordHash | Subscriptions
 *  - Posts:    ID | UserID | Заголовок | Текст | Теги | Приватность | Дата | Разрешенные пользователи
 *  - Comments: ID | PostID | UserID | Текст | Дата
 *  - Sessions: Token | UserID | CreatedAt
 *  - AccessRequests: ID | PostID | RequesterID | Status | Date
 */

function doGet(e) {
  try {
    const params = e.parameter || {};
    const action = params.action;

    let result;
    switch (action) {
      case 'register':        result = handleRegister(params); break;
      case 'login':           result = handleLogin(params); break;
      case 'addPost':         result = handleAddPost(params); break;
      case 'getPosts':        result = handleGetPosts(params); break;
      case 'editPost':        result = handleEditPost(params); break;
      case 'deletePost':      result = handleDeletePost(params); break;
      case 'subscribe':       result = handleSubscribe(params); break;
      case 'unsubscribe':     result = handleUnsubscribe(params); break;
      case 'addComment':      result = handleAddComment(params); break;
      case 'getComments':     result = handleGetComments(params); break;
      case 'getUserById':     result = handleGetUserById(params); break;
      case 'requestAccess':   result = handleRequestAccess(params); break;
      case 'approveAccess':   result = handleApproveAccess(params); break;
      case 'getAccessRequests': result = handleGetAccessRequests(params); break;
      case 'test':            result = { success: true, message: '✅ Сервер работает!' }; break;
      default:                result = { success: false, error: '❌ Неизвестное действие' };
    }

    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ==================== ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ ====================

function getSheet(name) {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
}

/**
 * Безопасная генерация нового ID — максимум по столбцу + 1.
 * Работает корректно даже после удаления строк.
 */
function getNextId(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return 1;
  const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues()
    .flat()
    .map(v => parseInt(v, 10))
    .filter(n => !isNaN(n));
  return ids.length ? Math.max(...ids) + 1 : 1;
}

/** SHA-256 хэш пароля в hex */
function hashPassword(password) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(password),
    Utilities.Charset.UTF_8
  );
  return bytes.map(b => ('0' + (b & 0xFF).toString(16)).slice(-2)).join('');
}

/** Валидация email */
function isValidEmail(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(String(email).trim());
}

/** Нормализация списка ID "2, 5,8" -> "2,5,8" */
function normalizeIdList(str) {
  return String(str || '')
    .split(',')
    .map(s => s.trim())
    .filter(Boolean)
    .join(',');
}

/** Преобразование строки подписок в массив */
function parseIdList(str) {
  return String(str || '')
    .split(',')
    .map(s => s.trim())
    .filter(Boolean);
}

// ==================== СЕССИИ ====================

function createSession(userId) {
  const sheet = getSheet('Sessions');
  if (!sheet) throw new Error('Таблица Sessions не найдена');
  const token = Utilities.getUuid();
  sheet.appendRow([token, String(userId), new Date().toISOString()]);
  return token;
}

function getUserIdByToken(token) {
  if (!token) return null;
  const sheet = getSheet('Sessions');
  if (!sheet) return null;
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(token)) {
      return String(data[i][1]);
    }
  }
  return null;
}

function destroySession(token) {
  if (!token) return;
  const sheet = getSheet('Sessions');
  if (!sheet) return;
  const data = sheet.getDataRange().getValues();
  for (let i = data.length - 1; i >= 1; i--) {
    if (String(data[i][0]) === String(token)) {
      sheet.deleteRow(i + 1);
    }
  }
}

// ==================== ПОЛЬЗОВАТЕЛИ ====================

function getUserByEmail(email) {
  const sheet = getSheet('Users');
  if (!sheet) return null;
  const data = sheet.getDataRange().getValues();
  const search = String(email).trim().toLowerCase();
  for (let i = 1; i < data.length; i++) {
    const userEmail = String(data[i][2] || '').trim().toLowerCase();
    if (userEmail === search) {
      return {
        id: String(data[i][0]),
        name: String(data[i][1] || ''),
        email: String(data[i][2] || ''),
        passwordHash: String(data[i][3] || ''),
        subscriptions: String(data[i][4] || '')
      };
    }
  }
  return null;
}

function getUserById(id) {
  const sheet = getSheet('Users');
  if (!sheet) return null;
  const data = sheet.getDataRange().getValues();
  const searchId = String(id);
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === searchId) {
      return {
        id: String(data[i][0]),
        name: String(data[i][1] || ''),
        email: String(data[i][2] || ''),
        subscriptions: String(data[i][4] || '')
      };
    }
  }
  return null;
}

// ==================== РЕГИСТРАЦИЯ ====================

function handleRegister(data) {
  const name = String(data.name || '').trim();
  const email = String(data.email || '').trim();
  const password = String(data.password || '');

  if (!name || !email || !password) {
    return { success: false, error: 'Все поля обязательны для заполнения' };
  }
  if (!isValidEmail(email)) {
    return { success: false, error: 'Некорректный email' };
  }
  if (password.length < 6) {
    return { success: false, error: 'Пароль должен быть не менее 6 символов' };
  }

  const sheet = getSheet('Users');
  if (!sheet) return { success: false, error: 'Таблица Users не найдена' };

  if (getUserByEmail(email)) {
    return { success: false, error: 'Пользователь с такой почтой уже существует' };
  }

  const newId = getNextId(sheet);
  const passwordHash = hashPassword(password);

  // ✅ ГЛАВНОЕ ИСПРАВЛЕНИЕ: реально сохраняем пользователя
  sheet.appendRow([newId, name, email, passwordHash, '']);

  const token = createSession(newId);

  return {
    success: true,
    token,
    user: { id: String(newId), name, email, subscriptions: '' }
  };
}

// ==================== ЛОГИН ====================

function handleLogin(data) {
  const email = String(data.email || '').trim();
  const password = String(data.password || '');

  if (!email || !password) {
    return { success: false, error: 'Все поля обязательны для заполнения' };
  }

  const user = getUserByEmail(email);
  if (!user) return { success: false, error: 'Пользователь не найден' };

  if (user.passwordHash !== hashPassword(password)) {
    return { success: false, error: 'Неверный пароль' };
  }

  const token = createSession(user.id);

  return {
    success: true,
    token,
    user: {
      id: user.id,
      name: user.name,
      email: user.email,
      subscriptions: user.subscriptions
    }
  };
}

// ==================== ПРОВЕРКА ТОКЕНА ====================

/**
 * Возвращает userId, если токен валиден.
 * Все защищённые операции используют его, а не data.userId.
 */
function requireAuth(data) {
  const token = data.token;
  const userId = getUserIdByToken(token);
  if (!userId) return null;
  return userId;
}

// ==================== ПОСТЫ ====================

function handleAddPost(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const title = String(data.title || '').trim();
  const content = String(data.content || '').trim();
  const tags = String(data.tags || '').trim();
  const isPrivate = data.isPrivate === 'true';
  const allowedUsers = normalizeIdList(data.allowedUsers);

  if (!title || !content) {
    return { success: false, error: 'Заголовок и текст поста обязательны' };
  }
  if (title.length > 200) return { success: false, error: 'Заголовок слишком длинный' };
  if (content.length > 10000) return { success: false, error: 'Текст слишком длинный' };

  const sheet = getSheet('Posts');
  if (!sheet) return { success: false, error: 'Таблица Posts не найдена' };

  const newId = getNextId(sheet);
  const date = new Date().toISOString();

  sheet.appendRow([
    newId,
    userId,
    title,
    content,
    tags,
    isPrivate ? 'true' : 'false',
    date,
    isPrivate ? allowedUsers : ''
  ]);

  return { success: true, postId: String(newId) };
}

function handleGetPosts(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const sheet = getSheet('Posts');
  if (!sheet) return { success: false, error: 'Таблица Posts не найдена' };

  const dataRows = sheet.getDataRange().getValues();
  const posts = [];
  const onlyPrivate = data.isPrivate === 'true';
  const onlyPublic  = data.isPrivate === 'false';

  // Предзагрузка подтверждённых запросов доступа для текущего пользователя
  const approvedPostIds = getApprovedPostIds(userId);

  for (let i = 1; i < dataRows.length; i++) {
    const row = dataRows[i];
    const postId = String(row[0]);
    const postUserId = String(row[1] || '');
    const privateFlag = String(row[5]) === 'true';

    if (onlyPrivate && !privateFlag) continue;
    if (onlyPublic && privateFlag) continue;

    let canView = false;
    if (!privateFlag) {
      canView = true;
    } else if (String(userId) === postUserId) {
      canView = true;
    } else {
      const allowedArray = parseIdList(row[7]);
      if (allowedArray.includes(String(userId)) || approvedPostIds.has(postId)) {
        canView = true;
      }
    }

    if (canView) {
      posts.push({
        id: postId,
        userId: postUserId,
        title: String(row[2] || ''),
        content: String(row[3] || ''),
        tags: String(row[4] || ''),
        isPrivate: privateFlag,
        createdAt: String(row[6] || ''),
        allowedUsers: parseIdList(row[7]).map(Number).filter(n => !isNaN(n))
      });
    }
  }

  posts.sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
  return { success: true, posts };
}

function handleEditPost(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const postId = String(data.postId || '');
  if (!postId) return { success: false, error: 'Не указан ID поста' };

  const title = String(data.title || '').trim();
  const content = String(data.content || '').trim();
  const tags = String(data.tags || '').trim();
  const isPrivate = data.isPrivate === 'true';
  const allowedUsers = normalizeIdList(data.allowedUsers);

  if (!title || !content) {
    return { success: false, error: 'Заголовок и текст обязательны' };
  }

  const sheet = getSheet('Posts');
  if (!sheet) return { success: false, error: 'Таблица Posts не найдена' };

  const dataRows = sheet.getDataRange().getValues();
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === postId) {
      if (String(dataRows[i][1]) !== String(userId)) {
        return { success: false, error: 'Нет прав для редактирования' };
      }
      sheet.getRange(i + 1, 3).setValue(title);
      sheet.getRange(i + 1, 4).setValue(content);
      sheet.getRange(i + 1, 5).setValue(tags);
      sheet.getRange(i + 1, 6).setValue(isPrivate ? 'true' : 'false');
      sheet.getRange(i + 1, 8).setValue(isPrivate ? allowedUsers : '');
      return { success: true };
    }
  }
  return { success: false, error: 'Пост не найден' };
}

function handleDeletePost(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const postId = String(data.postId || '');
  if (!postId) return { success: false, error: 'Не указан ID поста' };

  const sheet = getSheet('Posts');
  if (!sheet) return { success: false, error: 'Таблица Posts не найдена' };

  const dataRows = sheet.getDataRange().getValues();
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === postId) {
      if (String(dataRows[i][1]) !== String(userId)) {
        return { success: false, error: 'Нет прав для удаления' };
      }
      sheet.deleteRow(i + 1);
      return { success: true };
    }
  }
  return { success: false, error: 'Пост не найден' };
}

// ==================== ПОДПИСКИ ====================

function handleSubscribe(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const targetUserId = String(data.targetUserId || '').trim();
  if (!targetUserId) return { success: false, error: 'Не указан пользователь' };
  if (targetUserId === userId) return { success: false, error: 'Нельзя подписаться на себя' };

  const sheet = getSheet('Users');
  if (!sheet) return { success: false, error: 'Таблица Users не найдена' };

  const dataRows = sheet.getDataRange().getValues();
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === userId) {
      const subsArray = parseIdList(dataRows[i][4]);
      if (!subsArray.includes(targetUserId)) subsArray.push(targetUserId);
      sheet.getRange(i + 1, 5).setValue(subsArray.join(','));
      return { success: true, subscriptions: subsArray };
    }
  }
  return { success: false, error: 'Пользователь не найден' };
}

function handleUnsubscribe(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const targetUserId = String(data.targetUserId || '').trim();
  if (!targetUserId) return { success: false, error: 'Не указан пользователь' };

  const sheet = getSheet('Users');
  if (!sheet) return { success: false, error: 'Таблица Users не найдена' };

  const dataRows = sheet.getDataRange().getValues();
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][0]) === userId) {
      let subsArray = parseIdList(dataRows[i][4]);
      subsArray = subsArray.filter(id => id !== targetUserId);
      sheet.getRange(i + 1, 5).setValue(subsArray.join(','));
      return { success: true, subscriptions: subsArray };
    }
  }
  return { success: false, error: 'Пользователь не найден' };
}

// ==================== КОММЕНТАРИИ ====================

function handleAddComment(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const postId = String(data.postId || '').trim();
  const text = String(data.text || '').trim();
  if (!postId || !text) {
    return { success: false, error: 'Все поля комментария обязательны' };
  }
  if (text.length > 2000) {
    return { success: false, error: 'Комментарий слишком длинный' };
  }

  // Проверяем, что пост существует и доступен
  const access = canUserViewPost(userId, postId);
  if (!access) return { success: false, error: 'Нет доступа к посту' };

  const sheet = getSheet('Comments');
  if (!sheet) return { success: false, error: 'Таблица Comments не найдена' };

  const newId = getNextId(sheet);
  const date = new Date().toISOString();
  sheet.appendRow([newId, postId, userId, text, date]);

  return { success: true, commentId: String(newId) };
}

function canUserViewPost(userId, postId) {
  const sheet = getSheet('Posts');
  if (!sheet) return false;
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(postId)) {
      const postUserId = String(data[i][1]);
      const isPrivate = String(data[i][5]) === 'true';
      if (!isPrivate) return true;
      if (postUserId === String(userId)) return true;
      const allowed = parseIdList(data[i][7]);
      if (allowed.includes(String(userId))) return true;
      if (getApprovedPostIds(userId).has(String(postId))) return true;
      return false;
    }
  }
  return false;
}

function handleGetComments(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const postId = String(data.postId || '').trim();
  if (!postId) return { success: false, error: 'ID поста не указан' };

  if (!canUserViewPost(userId, postId)) {
    return { success: false, error: 'Нет доступа к посту' };
  }

  const sheet = getSheet('Comments');
  if (!sheet) return { success: false, error: 'Таблица Comments не найдена' };

  const dataRows = sheet.getDataRange().getValues();
  const comments = [];
  for (let i = 1; i < dataRows.length; i++) {
    if (String(dataRows[i][1]) === postId) {
      comments.push({
        id: String(dataRows[i][0]),
        postId: String(dataRows[i][1]),
        userId: String(dataRows[i][2]),
        text: String(dataRows[i][3] || ''),
        createdAt: String(dataRows[i][4] || '')
      });
    }
  }
  comments.sort((a, b) => new Date(a.createdAt) - new Date(b.createdAt));
  return { success: true, comments };
}

function handleGetUserById(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const targetId = String(data.userId || userId);

// ==================== ЗАПРОСЫ ДОСТУПА К ПРИВАТНЫМ ПОСТАМ ====================

function handleRequestAccess(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const postId = String(data.postId || '').trim();
  if (!postId) return { success: false, error: 'Не указан ID поста' };

  const sheet = getSheet('AccessRequests');
  if (!sheet) return { success: false, error: 'Таблица AccessRequests не найдена' };

  // Проверяем, нет ли уже заявки
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][1]) === postId && String(rows[i][2]) === userId) {
      if (String(rows[i][3]) === 'pending') {
        return { success: false, error: 'Заявка уже отправлена' };
      }
      if (String(rows[i][3]) === 'approved') {
        return { success: false, error: 'Доступ уже предоставлен' };
      }
    }
  }

  const newId = getNextId(sheet);
  sheet.appendRow([newId, postId, userId, 'pending', new Date().toISOString()]);
  return { success: true, requestId: String(newId) };
}

function handleApproveAccess(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const requestId = String(data.requestId || '').trim();
  if (!requestId) return { success: false, error: 'Не указан ID заявки' };

  const sheet = getSheet('AccessRequests');
  if (!sheet) return { success: false, error: 'Таблица AccessRequests не найдена' };

  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === requestId) {
      const postId = String(rows[i][1]);

      // Проверяем, что текущий пользователь — автор поста
      const postsSheet = getSheet('Posts');
      const postRows = postsSheet.getDataRange().getValues();
      let ownerId = null;
      for (let j = 1; j < postRows.length; j++) {
        if (String(postRows[j][0]) === postId) {
          ownerId = String(postRows[j][1]);
          break;
        }
      }
      if (ownerId !== userId) {
        return { success: false, error: 'Нет прав подтверждать заявку' };
      }

      sheet.getRange(i + 1, 4).setValue('approved');
      return { success: true };
    }
  }
  return { success: false, error: 'Заявка не найдена' };
}

function handleGetAccessRequests(data) {
  const userId = requireAuth(data);
  if (!userId) return { success: false, error: 'Не авторизован' };

  const postId = String(data.postId || '').trim();
  if (!postId) return { success: false, error: 'Не указан ID поста' };

  const sheet = getSheet('AccessRequests');
  if (!sheet) return { success: false, error: 'Таблица AccessRequests не найдена' };

  const rows = sheet.getDataRange().getValues();
  const requests = [];
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][1]) === postId) {
      requests.push({
        id: String(rows[i][0]),
        postId: String(rows[i][1]),
        requesterId: String(rows[i][2]),
        status: String(rows[i][3]),
        createdAt: String(rows[i][4] || '')
      });
    }
  }
  return { success: true, requests };
}

function getApprovedPostIds(userId) {
  const result = new Set();
  const sheet = getSheet('AccessRequests');
  if (!sheet) return result;
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][2]) === String(userId) && String(rows[i][3]) === 'approved') {
      result.add(String(rows[i][1]));
    }
  }
  return result;
}
